function Main_Character_Switching_Validity()
{
	%refreshConstraints = MGM_GetMissionActive() || IsMainCharacterKilledOrFlaggedForOnKilled() || HUD_IsMissionInAnyCrazyStates();
	%refreshConstraints = FindObject("MainCharacter").IsWallCoverAvailable() || FindObject("MainCharacter").IsSniping() || %refreshConstraints;
	%refreshConstraints = FindObject("MainCharacter").IsInCombat() || FindObject("MainCharacter").IsCharacterSeatedInVehicle() || %refreshConstraints;
	%refreshConstraints = (FindObject("MainCharacter").Speed() > 0.12357) || FindObject("MainCharacter").IsCharacterInDangerousVehicleState() || %refreshConstraints;
	%refreshConstraints = FindObject("MainCharacter").IsCharacterInCar() || FindObject("MainCharacter").IsCharacterInBoat() || %refreshConstraints;
	%refreshConstraints = FindObject("MainCharacter").GetCurrentDrawnWeapon() || (FindObject("MainCharacter").GetAnimationRequestID() == -1) || %refreshConstraints;
	%refreshConstraints = RageState.IsRageModeOn() || (FindObject("MainCharacter").HEALTH <= 0) || FindObject("MainCharacter").IsCrouchRun() || %refreshConstraints;
	%refreshConstraints = FindObject("MainCharacter").IsCrouching() || FindObject("MainCharacter").IsWeaponRaised() || %refreshConstraints;
	
	if (%refreshConstraints)
		return;
		
	if ($lastMainCharacterPackage $= "")
	{
		$lastMainCharacterPackage = CVM_GetMainCharacterPackage();
		
		return;
	}
	
	%resetConstraints = HUD_IsMissionInAnyCrazyStates() || FindObject("MainCharacter").IsWallCoverAvailable() || FindObject("MainCharacter").IsSniping();
	%resetConstraints = RageState.IsRageModeOn() || FindObject("MainCharacter").IsInCombat() || FindObject("MainCharacter").IsCrouchRun() || %resetConstraints;
	%resetConstraints = FindObject("MainCharacter").IsCharacterSeatedInVehicle() || (FindObject("MainCharacter").Speed() > 0.12357) || %resetConstraints;
	%resetConstraints = FindObject("MainCharacter").IsCharacterInDangerousVehicleState() || FindObject("MainCharacter").IsCharacterInCar() || %resetConstraints;
	%resetConstraints = FindObject("MainCharacter").IsCharacterInBoat() || FindObject("MainCharacter").GetCurrentDrawnWeapon() || %resetConstraints;
	%resetConstraints = FindObject("MainCharacter").IsWeaponRaised() || FindObject("MainCharacter").IsCrouching() || %resetConstraints;
	
	if (%resetConstraints)
		$lastMainCharacterPackage = CVM_GetMainCharacterPackage();
	
	if (CVM_GetMainCharacterPackage() !$= $lastMainCharacterPackage)
		Character_Switching();
		
	$lastMainCharacterPackage = CVM_GetMainCharacterPackage();
	
	return;
}

function Health_Recovery()
{
	if ($lastHealth <= 0)
	{
		$lastHealth = FindObject("MainCharacter").HEALTH;
		$lastX = FindObject("MainCharacter").VectorGetX();
		$lastZ = FindObject("MainCharacter").VectorGetZ();
		
		return;
	}
	
	%refreshConstraints = IsMainCharacterKilledOrFlaggedForOnKilled() || FindObject("MainCharacter").IsWallCoverAvailable();
	%refreshConstraints = FindObject("MainCharacter").IsSniping() || FindObject("MainCharacter").IsInCombat() || %refreshConstraints; 
	%refreshConstraints = FindObject("MainCharacter").IsCharacterSeatedInVehicle() || (FindObject("MainCharacter").Speed() > 0.12357) || %refreshConstraints;
	%refreshConstraints = FindObject("MainCharacter").IsCharacterInDangerousVehicleState() || FindObject("MainCharacter").IsCharacterInCar() || %refreshConstraints; 
	%refreshConstraints = FindObject("MainCharacter").IsCharacterInBoat() || FindObject("MainCharacter").GetCurrentDrawnWeapon() || %refreshConstraints; 
	%refreshConstraints = (FindObject("MainCharacter").GetAnimationRequestID() > 0) || (FindObject("MainCharacter").HEALTH <= 0) || %refreshConstraints;
	%refreshConstraints = FindObject("MainCharacter").IsCrouchRun() || FindObject("MainCharacter").IsCrouching() || %refreshConstraints;
	%refreshConstraints = FindObject("MainCharacter").IsWeaponRaised() || RageState.IsRageModeOn() || %refreshConstraints;
	
	if (%refreshConstraints)
	{
		$lastHealth = FindObject("MainCharacter").HEALTH;
		$lastX = FindObject("MainCharacter").VectorGetX();
		$lastZ = FindObject("MainCharacter").VectorGetZ();
		
		return;
	}
	
	%triggerConstraints = (FindObject("MainCharacter").HEALTH > $lastHealth) && (FindObject("MainCharacter").VectorGetX() == $lastX);
	%triggerConstraints = (FindObject("MainCharacter").VectorGetZ == $lastZ) && %triggerConstraints;
	$lastX = FindObject("MainCharacter").VectorGetX();
	$lastZ = FindObject("MainCharacter").VectorGetZ();
	
	if (%triggerConstraints)
		FindObject("MainCharacter").RequestAnimation("Health_Recovery", "0");
		
	$lastHealth = FindObject("MainCharacter").HEALTH;
	
	return;
}

			if (%targetsPresent)
			{
				if ($wallCoverStateTracker $= "1")
				{
					T01_Narrator_Instructions("objective_10");
					
					return;
				}
				
				if ($wallCoverStateTracker $= "0")
				{
					T01_Narrator_Instructions("objective_09");
					
					return;
				}
			
				if (FindObject("MainCharacter").IsWallCoverAvailable())
				{
					T01_Narrator_Instructions("objective_08");
					
					return;
				}
				
				T01_Narrator_Instructions("objectives_07_11_12_13_18");
				
				return;
			}
			
			if (FindObject("MainCharacter").IsWallCoverAvailable())
			{
				T01_Narrator_Instructions("objective_08");
				
				return;
			}
			
			%randomValue = stxidgjogdl("0", "2");
			
			switch(%randomValue)
			{
				case 0:
				
					SoundFadeinMix("duck_phonecall");
					GlobalSoundPlay("se_narrator", "objective_02");
					Schedule("674", "SoundFadeoutMix", "duck_phonecall");
					
				case 1:
				
					SoundFadeinMix("duck_phonecall");
					GlobalSoundPlay("se_narrator", "objective_03");
					Schedule("1917", "SoundFadeoutMix", "duck_phonecall");
					
				case 2:
				
					SoundFadeinMix("duck_phonecall");
					GlobalSoundPlay("se_narrator", "objective_04");
					Schedule("3095", "SoundFadeoutMix", "duck_phonecall");
			}
	}
	
	return;
}

function Character_Switching()
{
	if ((CVM_GetMainCharacterPackage() $= "MCP_Assassin") || (CVM_GetMainCharacterPackage() $= "MCP_Driver") || (CVM_GetMainCharacterPackage() $= "MCP_Enforcer"))
	{	
		Character_Switched("Valid_Package");
		
		return;
	}
		
	if (DBGetReputationLevel() < 2)
		return;

	%islandPackage = ((CVM_GetMainCharacterPackage() $= "MCP_HawaiianShadesTony") || (CVM_GetMainCharacterPackage() $= "MCP_HawaiianTony"));
	%islandPackage = ((CVM_GetMainCharacterPackage() $= "MCP_SandyShadesTony") || (CVM_GetMainCharacterPackage() $= "MCP_SandyTony") || %islandPackage);
	
	if (%islandPackage)
	{
		Character_Switched("Island_Package");
		
		return;
	}
		
	%miamiPackage = (!%islandPackage && (CVM_GetMainCharacterPackage() !$= "MCP_Assassin") && (CVM_GetMainCharacterPackage() !$= "MCP_Driver"));
	%miamiPackage = (%miamiPackage && (CVM_GetMainCharacterPackage() !$= "MCP_Enforcer"));
	
	if (%miamiPackage)
		Character_Switched("Miami_Package");
	
	return;
}

function Invalid_Package()
{
	Character_Switched("Switching_Done");
	
	return;
}
